cmake_minimum_required(VERSION 3.21)
project(MatterDesk VERSION 0.6.2 LANGUAGES CXX)
set(MATTERCAMVIEWER_VERSION 2.4.2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network WebSockets WebEngineWidgets WebChannel QuickWidgets)
add_library(mattercore src/matterclient.cpp src/matterdevice.cpp src/homestore.cpp src/camerametadata.cpp)
target_include_directories(mattercore PUBLIC src)
target_link_libraries(mattercore PUBLIC Qt6::Core Qt6::Network Qt6::WebSockets)
target_compile_options(mattercore PRIVATE -Wall -Wextra -Wpedantic)
add_library(matterwidgets OBJECT src/accessorypanel.cpp src/homeeditors.cpp src/mainwindow.cpp src/devicecard.cpp src/miniwindow.cpp src/snapshotpreview.cpp src/camerawindow.cpp src/camerasnapshots.cpp src/cameraptzcontrols.cpp src/camerametadatapanel.cpp src/camera/camera.qrc)
target_link_libraries(matterwidgets PUBLIC mattercore Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::QuickWidgets)
target_compile_options(matterwidgets PRIVATE -Wall -Wextra -Wpedantic)
add_executable(matterdesk src/main.cpp)
target_link_libraries(matterdesk PRIVATE matterwidgets)
target_compile_options(matterdesk PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mattercamviewer src/mini_main.cpp)
target_link_libraries(mattercamviewer PRIVATE matterwidgets)
target_compile_options(mattercamviewer PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(mattercamviewer PRIVATE MATTERCAMVIEWER_VERSION="${MATTERCAMVIEWER_VERSION}")
# Keep the previous build target and launcher path available during migration.
add_custom_target(matterdesk-mini DEPENDS mattercamviewer)
add_custom_command(TARGET mattercamviewer POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E create_symlink "$<TARGET_FILE_NAME:mattercamviewer>" "$<TARGET_FILE_DIR:mattercamviewer>/matterdesk-mini")
# Opt-in synthetic PTZ / OHF integration probe; never part of CTest.
add_executable(ptz_ws_probe EXCLUDE_FROM_ALL tests/ptz_ws_probe.cpp)
target_link_libraries(ptz_ws_probe PRIVATE mattercore)
add_executable(ptz_widget_probe EXCLUDE_FROM_ALL tests/ptz_widget_probe.cpp)
target_link_libraries(ptz_widget_probe PRIVATE matterwidgets)
